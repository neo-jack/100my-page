async function tintDesignImage(imagePath, mode = "auto") {
  const { data, info } = await (0, import_sharp.default)(imagePath).ensureAlpha().raw().toBuffer({ resolveWithObject: true });
  const pixels = new Uint8Array(data.buffer);
  const resolvedMode = mode === "auto" ? detectMode(pixels) : mode;

  }
  if (resolvedMode === "difference") {

function detectMode(pixels) {
  let brightCount = 0;
  const sampleSize = Math.min(pixels.length / 4, 1e3);
  const step = Math.floor(pixels.length / 4 / sampleSize);
  for (let i = 0; i < sampleSize; i++) {
    const idx = i * step * 4;
    const brightness = (pixels[idx] + pixels[idx + 1] + pixels[idx + 2]) / 3;
    if (brightness > 220)
      brightCount++;
  }
  return brightCount / sampleSize > 0.5 ? "magenta" : "ghost";
}
function tintMagenta(pixels, info) {
  for (let i = 0; i < pixels.length; i += 4) {
    const r = pixels[i], g = pixels[i + 1], b = pixels[i + 2];
    const brightness = (r + g + b) / 3;
    if (brightness > LIGHT_THRESHOLD) {
      pixels[i + 3] = 0;
    } else {
      const darkness = 1 - brightness / 255;
      pixels[i] = MAGENTA[0];
      pixels[i + 1] = MAGENTA[1];
      pixels[i + 2] = MAGENTA[2];
      pixels[i + 3] = Math.round(darkness * 200);
    }
  }
  return (0, import_sharp.default)(Buffer.from(pixels.buffer), {
    raw: { width: info.width, height: info.height, channels: 4 }
  }).png().toBuffer();
}
function tintGhost(pixels, info, opacity) {
  for (let i = 0; i < pixels.length; i += 4) {
    pixels[i + 3] = Math.round(pixels[i + 3] * opacity);
  }
  return (0, import_sharp.default)(Buffer.from(pixels.buffer), {
    raw: { width: info.width, height: info.height, channels: 4 }
  }).png().toBuffer();
}

  const q = model.border?.quad ?? model.content.quad;

  const maxX = Math.max(q[0], q[2], q[4], q[6]);

  const y = Math.max(0, minY - padding);

  return { x, y, width, height };

  return PlaywrightEngine2.create(options.url, {
    headless: options.headless ?? true,
    viewport: options.viewport
  });
}

// engine/selector.js
function resolveSelector(input) {
  if (input.startsWith("~")) {
    const keyword = input.slice(1).trim();
    return `[class*="${keyword}"]`;
  }
  return input;
}

// commands/overlay.js
function buildOverlayOptions(raw) {
  if (!raw.design)
    throw new Error("--design is required (path to design screenshot)");
  if (!raw.url)
    throw new Error("--url is required (target page URL)");
  return {
    designImagePath: (0, import_path.resolve)(raw.design),
    targetUrl: raw.url,
    cdp: raw.cdp,
    output: raw.output,
    selector: raw.selector ? resolveSelector(raw.selector) : void 0,
    offsetX: raw.offsetX != null ? parseFloat(raw.offsetX) : void 0,
    offsetY: raw.offsetY != null ? parseFloat(raw.offsetY) : void 0,
    scale: raw.scale != null ? parseFloat(raw.scale) : void 0,
    opacity: raw.opacity != null ? parseFloat(raw.opacity) : void 0,
    fullPage: raw.fullPage ?? false
  };
}
async function captureGhost(opts, params) {
  const engine = await createEngine({ url: opts.targetUrl, cdp: opts.cdp });
  try {
    await engine.injectOverlay({
      designImagePath: opts.designImagePath,
      targetUrl: opts.targetUrl,
      offsetX: params.offsetX,
      offsetY: params.offsetY,
      scale: params.scale,
      opacity: params.opacity
    });
    const buf = await engine.captureOverlay({ fullPage: opts.fullPage });
    const outputPath = opts.output ?? `overlay-${Date.now()}.png`;
    await (0, import_promises.mkdir)((0, import_path.dirname)((0, import_path.resolve)(outputPath)), { recursive: true }).catch(() => {
    });
    await (0, import_promises.writeFile)(outputPath, buf);
    console.log(JSON.stringify({ output: outputPath, bytes: buf.length }));
  } finally {
    await engine.close();
  }
}
async function compositeGhost(opts) {
  const sharp2 = (await import("sharp")).default;
  const { tintDesignImage: tintDesignImage2 } = await Promise.resolve().then(() => (init_tint(), tint_exports));
  const engine = await createEngine({ url: opts.targetUrl, cdp: opts.cdp });
  let elementBuf;
  try {
    elementBuf = await engine.screenshot({ selector: opts.selector, fullPage: opts.fullPage });
  } finally {
    await engine.close();
  }
  const elementMeta = await sharp2(elementBuf).metadata();
  const ew = elementMeta.width;
  const eh = elementMeta.height;
  const tintedBuf = await tintDesignImage2(opts.designImagePath);
  const tintedResized = await sharp2(tintedBuf).resize(ew, eh, { fit: "contain", background: { r: 0, g: 0, b: 0, alpha: 0 } }).toBuffer();
  const ghostBuf = await sharp2(elementBuf).composite([{ input: tintedResized, blend: "over" }]).png().toBuffer();
  const outputPath = opts.output ?? `overlay-${Date.now()}.png`;
  await (0, import_promises.mkdir)((0, import_path.dirname)((0, import_path.resolve)(outputPath)), { recursive: true }).catch(() => {
  });
  await (0, import_promises.writeFile)(outputPath, ghostBuf);
  console.log(JSON.stringify({ output: outputPath, bytes: ghostBuf.length, selector: opts.selector, elementSize: { width: ew, height: eh } }));
}
async function overlay(raw) {
  const opts = buildOverlayOptions(raw);
  if (opts.selector) {
    await compositeGhost(opts);
    return;
  }
  if (opts.offsetX != null || opts.offsetY != null) {
    await captureGhost(opts, {
      offsetX: opts.offsetX ?? 0,
      offsetY: opts.offsetY ?? 0,
      scale: opts.scale ?? 1,
      opacity: opts.opacity ?? 1
    });
    return;
  }
  throw new Error("overlay requires --selector for component comparison, or --offset-x/--offset-y for full-page mode");
}
